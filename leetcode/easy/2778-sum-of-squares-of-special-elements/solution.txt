class Solution {
public:
    int sumOfSquares(vector<int>& nums) {
        int n = nums.size();
        int  sum=0;
        for(int i=1;i<=n;i++){
            if(n%i==0){
               int  product = nums[i-1]*nums[i-1]; 
                // c++ mein indexing vector ki 0 se hoti hai isliye nums[i] par ye error de dega 
                sum +=product;  
            }
            
        }
        return sum;
    }
};
