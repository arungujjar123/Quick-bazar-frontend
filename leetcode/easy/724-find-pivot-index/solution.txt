        // optimised solution hai iska joki prefix sum se hoga like in o(n) time complexity mein

        for(int num:nums){
            totalsum +=num;
        }
           int leftsum=0;
        for(int i=0;i<n;i++){
            int rightsum = totalsum-leftsum-nums[i];
            if(leftsum==rightsum){
                return i;
            }else{
                leftsum +=nums[i];
