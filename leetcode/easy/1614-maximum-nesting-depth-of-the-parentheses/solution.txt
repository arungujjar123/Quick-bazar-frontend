class Solution {
public:
    int maxDepth(string s) {
        int n = s.size();
        int maxim = 0;
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == '(') {
                count++;

            } else if (s[i] == ')') {
                count--;
            }
            maxim = max(maxim, count);
        }
        return maxim;
    }
};
