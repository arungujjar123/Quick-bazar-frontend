        // prev = currsum - k
            int prev = currsum - k;

            // Step 3: Check if required previous prefix
            // sum exists in HashMap
            if(mp.find(prev) != mp.end()) {

                // Add its frequency because multiple
                // subarrays can have the same sum
                count += mp[prev];
            }

            // Step 4: Store/update current prefix sum
            // in HashMap for future subarrays
            mp[currsum]++;
        }

        // Step 5: Return total valid subarrays
        return count;
    }
};
