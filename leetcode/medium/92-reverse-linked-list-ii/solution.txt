r->next;

            // Curr ka next, temp ke next se connect karo
            // Isse temp current position se nikal jayega
            curr->next = temp->next;

            // Temp ko reversed part ke starting node se jodo
            temp->next = prev->next;

            // Prev ka next temp kar do
            // Ab temp reversed part ke start mein aa jayega
            prev->next = temp;
        }

        // Dummy ke next se actual reversed list milegi
        return dummy->next;
    }
};
