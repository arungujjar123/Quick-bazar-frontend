nough budget -> every difference becomes 0
        if (sum <= k) return 0;

        // Step 2: shave the biggest differences, level by level
        for (int i = mx; i > 0 && k > 0; i--) {
            long long move = min(k, (long long)d[i]);
            d[i] -= move;
            d[i - 1] += move;
            k -= move;
        }

        // Step 3: add up the squares
        long long ans = 0;
        for (int i = 0; i <= mx; i++)
            ans += (long long)i * i * d[i];

        return ans;
    }
