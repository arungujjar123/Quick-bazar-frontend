class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        // ==================== optimised o(1)
        int n=nums.size();
        vector<int>answer(n,1);
        
        // pass 1  har index ka left-side product store karo 
        int prefix=1;

        for(int i=0;i<n;i++){
            answer[i] = prefix;
            prefix *=nums[i];
        }
        // pass 2
        int suffix = 1;
        for(int i=n-1;i>=0;i--){
            answer[i] *=suffix;
            suffix *=nums[i];
        }

        return answer;
