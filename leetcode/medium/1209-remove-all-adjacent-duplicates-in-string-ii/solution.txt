class Solution {
public:
    string removeDuplicates(string s, int k) {
        // brute force
        int n = s.size();
        int i=0;
        while(i<(int)s.size()){
             
             int count=1;
             while(i+count < (int)s.size() && s[i] == s[i+count]){
                count++;
             }
            
            if(count>=k){
                s.erase(i,k);

                i = max(0,i-k+1);
            }else{
                i++;
            }
        }
        return s;
    }
};
