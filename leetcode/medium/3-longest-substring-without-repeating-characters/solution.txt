{
        // dynamic slidining window mein two element maintain karo
        // exapand par right wala badao if condition satisfie
        // shrink left badao if condition is not satisfied
        int left = 0;
        int ans = 0;

        unordered_set<char> st;

        for (int right = 0; right < s.size(); right++) {
            // duplicate hai toh window shirnk karo
            while(st.find(s[right]) != st.end()) {
                st.erase(s[left]);
                left++;
            }

            // current character add karo 
            st.insert(s[right]);

            // current valide window ki length 
             ans = max(ans,right-left+1);
        }
        return ans;
