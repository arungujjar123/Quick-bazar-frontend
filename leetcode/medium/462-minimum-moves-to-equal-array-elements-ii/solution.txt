class Solution {
public:
    int minMoves2(vector<int>& nums) {
        int n = nums.size();
        long long moves = LLONG_MAX;

        for (int i = 0; i < n; i++) {
            long long currMoves = 0;
            int target = nums[i];

            for (int j = 0; j < n; j++) {
                currMoves += abs((long long)target - nums[j]);
            }

            moves = min(moves, currMoves);
        }

        return moves;
    }
};
