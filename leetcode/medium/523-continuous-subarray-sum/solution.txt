; i < n; i++) {
            sum += nums[i];

            int rem = sum % k;

            // Agar same remainder pehle aa chuka hai
            if (mp.find(rem) != mp.end()) {

                // Subarray ki length kam se kam 2 honi chahiye
                if (i - mp[rem] >= 2) {
                    return true;
                }

            } else {
                // Sirf pehli baar aane wala index store karo
                mp[rem] = i;
            }
        }

        return false;
    }
};
